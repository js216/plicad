# Each lid comes 30 mm down onto the one below and touches it; the file
# samples the paths every 0.25 mm, the second lid every 2 mm, and the
# plates of 2 mm hold that to 1 mm.
step 0.25 1
clearance 0.2
part base ../build/plate.st
part lid ../build/plate.st move 0 0 2 path move 0 0 -30
part top ../build/plate.st move 0 0 4 path move 0 0 -30 step 2 5
